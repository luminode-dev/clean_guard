# Jetson (JetPack 6.x) 기준.
# 주의: torch/torchvision은 pip 기본 wheel이 아니라 NVIDIA Jetson용 wheel을 설치할 것
#   https://docs.ultralytics.com/guides/nvidia-jetson/ 참고
ultralytics>=8.4          # --backend ultralytics(기본)용. --backend trt만 쓰면 torch/ultralytics 불필요
lap>=0.5.12               # ultralytics ByteTrack 의존성
numpy<2                   # JetPack cv2가 numpy 1.x ABI로 빌드됨. 2.x면 cv2가 깨진다
scipy                     # bytetrack.py (헝가리안 매칭)

# opencv-python>=4.8      # Jetson에서는 설치하지 말 것
#   pip 배포판은 GStreamer 없이(FFMPEG only) 빌드돼 있어, 설치하면 JetPack 기본
#   cv2(4.8.0, GStreamer 1.20.3)를 가려 RTSP가 소프트웨어 디코딩으로 떨어진다.
#   ultralytics가 의존성으로 끌어오므로 설치 후 `pip uninstall opencv-python` 필요.

# --backend trt (torch 없는 경로) — Jetson:
#   tensorrt      JetPack 기본 포함 (python3-libnvinfer)
#   cuda-python   pip3 install cuda-python   (GPU 버퍼 관리)
#   PC 개발용 .onnx 실행에는 onnxruntime — 단 Jetson 기본 cv2와 numpy 충돌하므로 보드에선 .engine만 쓸 것

# --- TTS 방송(--tts) 의존성은 여기에 넣지 말 것 ---
# onnxruntime이 numpy 2.x를 끌어오는데 JetPack 기본 cv2는 numpy 1.x로 빌드되어 있어
# 같은 환경에 섞으면 cv2가 깨진다. TTS 의존성(supertonic / onnxruntime / soundfile /
# huggingface-hub)은 setup_tts.sh가 격리된 .venv-tts/ 에 설치한다.
# 감시 프로세스는 사전 렌더링된 wav를 paplay로 재생할 뿐이라 추가 런타임 의존성이 없다.
